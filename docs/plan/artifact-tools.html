Key code

reigner/tools/artifacts/store.py — the trust boundary
class ArtifactStore:
    def __init__(self, root: str | Path, schema: ArtifactSchema) -> None:
        self.root = Path(root).resolve()
        self.schema = schema

    def resolve(self, rel: str) -> Path:
        """Resolve a root-relative path, rejecting traversal escapes."""
        if not rel or rel.startswith(("/", "\\")):
            raise ValueError(f"path must be relative, got {rel!r}")
        candidate = (self.root / rel).resolve()
        try:
            candidate.relative_to(self.root)
        except ValueError as exc:
            raise ValueError(f"path {rel!r} escapes artifact root") from exc
        return candidate

    def tools(self) -> list:
        from .read import build_read_artifact_file
        from .grep import build_grep_artifact
        from .json_field import build_get_json_field
        from .list import build_list_documents, build_list_versions, build_get_section
        return [
            build_read_artifact_file(self),
            build_grep_artifact(self),
            build_get_json_field(self),
            build_list_documents(self),
            build_list_versions(self),
            build_get_section(self),
        ]
reigner/tools/artifacts/list.py — signatures derived from schema.entity_path_keys()
def build_list_versions(store: ArtifactStore):
    keys = store.schema.entity_path_keys()        # e.g. ("ticker", "fiscal_year")
    *prefix_keys, trailing = keys

    # Synthesize an async fn whose signature matches the prefix keys exactly,
    # so @tool's pydantic JSON-schema builder sees the correct argument names.
    params = [
        inspect.Parameter(k, inspect.Parameter.KEYWORD_ONLY, annotation=str)
        for k in prefix_keys
    ]
    sig = inspect.Signature(parameters=params, return_annotation=list[str])

    async def list_versions(**identifiers: str) -> list[str]:
        """List the values of the trailing entity_path placeholder for a prefix."""
        prefix_dir = store.resolve(
            "/".join(identifiers[k] for k in prefix_keys)
        )
        if not prefix_dir.is_dir():
            return []
        return sorted(p.name for p in prefix_dir.iterdir() if p.is_dir())

    list_versions.__signature__ = sig
    list_versions.__doc__ = f"List {trailing} values for a given {', '.join(prefix_keys)}."
    return tool(readonly=True)(list_versions)
reigner/tools/artifacts/read.py — bounded, self-describing result
def build_read_artifact_file(store: ArtifactStore):
    @tool(readonly=True)
    async def read_artifact_file(
        path: str, offset: int = 0, limit: int = 4000
    ) -> dict:
        """Read a chunk of a text artifact under the artifact root."""
        target = store.resolve(path)
        if not target.is_file():
            raise FileNotFoundError(f"no artifact at {path!r}")
        if not is_text_file(target, store.schema):
            raise ValueError(f"{path!r} is not a readable text artifact")
        text = target.read_text()
        chunk = text[offset : offset + limit]
        return {
            "content": chunk,
            "offset": offset,
            "limit": limit,
            "has_more": offset + limit < len(text),
            "total_size": len(text),
        }
    return read_artifact_file
